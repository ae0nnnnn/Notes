## Binary Search

```
item = ["Help", "He", "ddf", "333", "ff"]
found = false 
first = 0
last = item.length - 1
target = "ddf"
midpoint = 0


while first <= last AND found == false 
	midpoint = (first + last) DIV 2
	if item[midpoint] == target then
		found = true 
	else
		if item[midpoint] < target then
			first = midpoint + 1
		else 
			last = midpoint -1
		endif 
	endif 
endwhile 

if found == true  then
	print("Found " + target + " at "   midpoint)
endif
```


## Bubble Sort

```
item = ["1", "5", "6", "8", "6"]
n = item.length 
swapped = true 

while n > 0 AND swapped
	swapped = false
	n = n - 1
	for index = 0 to (n - 1)
		if item[index] > item[index + 1] then
			temp = item[index]
			item[index] = item[index + 1]
			item[index + 1] = temp
		endif 	
	next index
endwhile 


print(item)
```


## Insertion Sort

```
items = ["999999", "5", "6", "8", "6"]

for index = 1 to items.length -1
	current = items[index]
	index2 = index
	while index2 > 0 AND items[index2 -1] > current 
		items[index2] = items[index2 -1]
		index2 = index2 -1
	endwhile 

	items[index2] = current 
next index

print(items)
```

## Merge Sort

```
items = ["999999", "5", "6", "8", "6"]

function merge(list1, list2)
    newlist = []
    i1 = 0

            newlist.append(list2[i2])
            i2 = i2 + 1
        elseif list1[i1] < list2[i2] then
            newlist.append(list1[i1])
            i1 = i1 + 1
        else
            newlist.append(list1[i1])
            newlist.append(list2[i2])
            i1 = i1 + 1
            i2 = i2 + 1
        endif
    endwhile

        newlist.append(list1[i1])
        i1 = i1 + 1
    endwhile
    while i2 < list2.length
        newlist.append(list2[i2])
        i2 = i2 + 1
    endwhile

    return newlist
endfunction
```
